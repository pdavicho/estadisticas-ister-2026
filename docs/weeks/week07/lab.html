Análisis de Datos Reales con RStudio: Desigualdad Salarial en Ecuador

Autor/a

Ing. David Minango PhD

PRÁCTICA EXPERIMENTAL APP 2 | Duración: 8 horas

Esta práctica corresponde a la segunda evaluación experimental del curso (10 puntos Parcial 2). Analizará datos reales de desigualdad salarial en Ecuador usando R y Python, presentando conclusiones con impacto social.

Contexto y Problemática

La brecha salarial es uno de los principales problemas de equidad en Ecuador y América Latina. Según el INEC (Instituto Nacional de Estadística y Censos), existen diferencias significativas de ingresos por género, región geográfica, nivel educativo y sector (público/privado/informal).

En esta práctica, usted asumirá el rol de analista de datos para el Ministerio de Trabajo, con el objetivo de cuantificar y visualizar estas brechas usando herramientas estadísticas.

Instrucciones Generales

  1. Trabaje en RStudio (o Google Colab con R) para el análisis principal
  2. Complemente con Python para la visualización final
  3. Documente cada paso con comentarios en el código
  4. Interprete todos los resultados en el contexto de la problemática social
  5. Presente sus hallazgos en un informe ejecutivo

Fase 1 — Obtención y Carga del Dataset

Descargue el dataset desde Moodle o genérelo con el siguiente código:

# ══════════════════════════════════════════════════════════════
# APP 2 — Análisis de Desigualdad Salarial Ecuador
# Herramientas Estadísticas | ISTER 2026
# Estudiante: [SU NOMBRE]
# Fecha: [FECHA]
# ══════════════════════════════════════════════════════════════

library(dplyr)
library(ggplot2)  # si está disponible

set.seed(2026)
n <- 300

datos_ecuador <- data.frame(
  id        = 1:n,
  region    = sample(c("Costa","Sierra","Oriente","Insular"), n,
                     replace=TRUE, prob=c(0.43, 0.44, 0.11, 0.02)),
  provincia = sample(c("Pichincha","Guayas","Azuay","Manabí",
                        "Tungurahua","Otras"), n, replace=TRUE),
  sector    = sample(c("Público","Privado Formal","Informal"), n,
                     replace=TRUE, prob=c(0.28, 0.42, 0.30)),
  genero    = sample(c("Hombre","Mujer"), n, replace=TRUE, prob=c(0.51, 0.49)),
  nivel_edu = factor(
    sample(c("Sin instrucción","Primaria","Secundaria","Superior","Postgrado"), n,
           replace=TRUE, prob=c(0.05, 0.20, 0.35, 0.32, 0.08)),
    levels=c("Sin instrucción","Primaria","Secundaria","Superior","Postgrado"),
    ordered=TRUE
  ),
  edad      = sample(18:65, n, replace=TRUE),
  anios_exp = sample(0:40, n, replace=TRUE),
  horas_sem = sample(20:60, n, replace=TRUE),
  salario   = round(pmax(400, rnorm(n,
    mean = 850 + 200*(as.numeric(factor(sample(c("Hombre","Mujer"), n, TRUE, c(.51,.49)))) - 1.5) +
           150*(as.numeric(factor(sample(c("Público","Privado Formal","Informal"), n, TRUE, c(.28,.42,.30)))) - 2),
    sd = 380)))
)

# Introducir valores faltantes realistas
set.seed(99)
datos_ecuador$salario[sample(1:n, 15)] <- NA
datos_ecuador$nivel_edu[sample(1:n, 8)] <- NA
datos_ecuador$horas_sem[sample(1:n, 5)] <- NA

# Guardar para uso posterior
write.csv(datos_ecuador, "datos_salarios_ecuador_app2.csv", row.names=FALSE)

cat("✅ Dataset cargado\n")
cat("Dimensiones:", dim(datos_ecuador), "\n")
cat("NAs por columna:\n")
print(colSums(is.na(datos_ecuador)))

Fase 3 — Análisis Estadístico Descriptivo

# 3.1 Estadísticas globales del salario
cat("=== ESTADÍSTICAS GLOBALES ===\n")
n_valido    <- nrow(datos_limpios)
media_sal   <- mean(datos_limpios$salario)
mediana_sal <- median(datos_limpios$salario)
desv_sal    <- sd(datos_limpios$salario)
cv_sal      <- desv_sal / media_sal * 100
asimetria   <- (3 * (media_sal - mediana_sal)) / desv_sal  # asimetría de Pearson

cat(sprintf("n = %d\n", n_valido))
cat(sprintf("Media:   $%.2f\n", media_sal))
cat(sprintf("Mediana: $%.2f\n", mediana_sal))
cat(sprintf("Moda (aprox.): mayor frecuencia en histograma\n"))
cat(sprintf("Desv. Estándar: $%.2f\n", desv_sal))
cat(sprintf("CV: %.2f%%\n", cv_sal))
cat(sprintf("Coef. Asimetría (Pearson): %.4f\n", asimetria))
cat(sprintf("  → %s\n", ifelse(asimetria > 0, "Asimetría positiva (cola a la derecha)", "Asimetría negativa")))

# 3.2 Distribución por cuartiles
cat("\n=== DISTRIBUCIÓN POR CUARTILES ===\n")
print(quantile(datos_limpios$salario, c(0,0.25,0.50,0.75,0.90,0.95,1)))

# 3.3 Por región
cat("\n=== ANÁLISIS POR REGIÓN ===\n")
resumen_region <- datos_limpios %>%
  group_by(region) %>%
  summarise(
    n = n(),
    media = round(mean(salario)),
    mediana = median(salario),
    cv = round(sd(salario)/mean(salario)*100, 1),
    q1 = quantile(salario, 0.25),
    q3 = quantile(salario, 0.75)
  ) %>%
  arrange(desc(media))
print(as.data.frame(resumen_region))

# 3.4 Brecha de género (indicador clave)
cat("\n=== BRECHA SALARIAL DE GÉNERO ===\n")
por_genero <- datos_limpios %>%
  group_by(genero) %>%
  summarise(media = round(mean(salario)), mediana = median(salario), n = n())
print(as.data.frame(por_genero))

media_h <- por_genero$media[por_genero$genero=="Hombre"]
media_m <- por_genero$media[por_genero$genero=="Mujer"]
brecha_abs <- media_h - media_m
brecha_pct <- brecha_abs / media_h * 100

cat(sprintf("Brecha absoluta: $%.0f\n", brecha_abs))
cat(sprintf("Brecha relativa: %.1f%% (las mujeres ganan %.1f%% menos)\n", brecha_pct, brecha_pct))

# 3.5 Por sector
cat("\n=== ANÁLISIS POR SECTOR ===\n")
por_sector <- datos_limpios %>%
  group_by(sector) %>%
  summarise(n=n(), media=round(mean(salario)), mediana=median(salario), cv=round(sd(salario)/mean(salario)*100,1)) %>%
  arrange(desc(media))
print(as.data.frame(por_sector))

# 3.6 Por nivel educativo
cat("\n=== SALARIO POR NIVEL EDUCATIVO ===\n")
por_edu <- datos_limpios %>%
  group_by(nivel_edu) %>%
  summarise(n=n(), media=round(mean(salario)), mediana=median(salario)) %>%
  arrange(nivel_edu)
print(as.data.frame(por_edu))

# 3.7 Correlaciones
cat("\n=== CORRELACIONES ===\n")
cor_sal_exp  <- cor(datos_limpios$salario, datos_limpios$anios_exp)
cor_sal_edad <- cor(datos_limpios$salario, datos_limpios$edad)
cor_sal_hrs  <- cor(datos_limpios$salario, datos_limpios$horas_sem)
cat(sprintf("Salario ~ Experiencia: r = %.4f\n", cor_sal_exp))
cat(sprintf("Salario ~ Edad:        r = %.4f\n", cor_sal_edad))
cat(sprintf("Salario ~ Horas/sem:   r = %.4f\n", cor_sal_hrs))

Fase 5 — Análisis Final en Python

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Cargar datos
df = pd.read_csv("datos_salarios_ecuador_app2.csv")
df.dropna(inplace=True)

print("Dataset:", df.shape)
print("\nMedia por región:")
print(df.groupby('region')['salario'].mean().sort_values(ascending=False).round(2))

print("\nBrecha de género:")
por_genero = df.groupby('genero')['salario'].agg(['mean','median'])
print(por_genero.round(2))

# Visualización final con Python
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
fig.suptitle('Análisis de Desigualdad Salarial — Ecuador 2023', fontsize=14, fontweight='bold')

# Histograma
axes[0,0].hist(df['salario'], bins=20, color='#0055a5', edgecolor='white')
axes[0,0].axvline(df['salario'].mean(), color='red', lw=2, linestyle='--', label='Media')
axes[0,0].axvline(df['salario'].median(), color='green', lw=2, linestyle='--', label='Mediana')
axes[0,0].set_title('Distribución Salarial')
axes[0,0].legend()

# Brecha género
genero_media = df.groupby('genero')['salario'].mean()
axes[0,1].bar(genero_media.index, genero_media.values,
              color=['#4a90d9','#e84393'], edgecolor='white')
axes[0,1].set_title('Salario Medio por Género')
axes[0,1].set_ylabel('USD')

# Por sector
sector_media = df.groupby('sector')['salario'].mean().sort_values()
axes[1,0].barh(sector_media.index, sector_media.values, color='#28a745')
axes[1,0].set_title('Salario por Sector')
axes[1,0].set_xlabel('USD')

# Por nivel educativo
edu_order = ['Sin instrucción','Primaria','Secundaria','Superior','Postgrado']
edu_media = df.groupby('nivel_edu')['salario'].mean()
edu_media = edu_media.reindex([e for e in edu_order if e in edu_media.index])
axes[1,1].bar(range(len(edu_media)), edu_media.values,
              color='#ffc107', edgecolor='white')
axes[1,1].set_xticks(range(len(edu_media)))
axes[1,1].set_xticklabels(edu_media.index, rotation=25, ha='right', fontsize=8)
axes[1,1].set_title('Retorno de la Educación')
axes[1,1].set_ylabel('Salario Medio (USD)')

plt.tight_layout()
plt.savefig('analisis_salarial_ecuador.png', dpi=150, bbox_inches='tight')
plt.show()
print("✅ Gráfico guardado")

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Cargar datos
df = pd.read_csv("datos_salarios_ecuador_app2.csv")
df.dropna(inplace=True)

print("Dataset:", df.shape)
print("\nMedia por región:")
print(df.groupby('region')['salario'].mean().sort_values(ascending=False).round(2))

print("\nBrecha de género:")
por_genero = df.groupby('genero')['salario'].agg(['mean','median'])
print(por_genero.round(2))

# Visualización final con Python
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
fig.suptitle('Análisis de Desigualdad Salarial — Ecuador 2023', fontsize=14, fontweight='bold')

# Histograma
axes[0,0].hist(df['salario'], bins=20, color='#0055a5', edgecolor='white')
axes[0,0].axvline(df['salario'].mean(), color='red', lw=2, linestyle='--', label='Media')
axes[0,0].axvline(df['salario'].median(), color='green', lw=2, linestyle='--', label='Mediana')
axes[0,0].set_title('Distribución Salarial')
axes[0,0].legend()

# Brecha género
genero_media = df.groupby('genero')['salario'].mean()
axes[0,1].bar(genero_media.index, genero_media.values,
              color=['#4a90d9','#e84393'], edgecolor='white')
axes[0,1].set_title('Salario Medio por Género')
axes[0,1].set_ylabel('USD')

# Por sector
sector_media = df.groupby('sector')['salario'].mean().sort_values()
axes[1,0].barh(sector_media.index, sector_media.values, color='#28a745')
axes[1,0].set_title('Salario por Sector')
axes[1,0].set_xlabel('USD')

# Por nivel educativo
edu_order = ['Sin instrucción','Primaria','Secundaria','Superior','Postgrado']
edu_media = df.groupby('nivel_edu')['salario'].mean()
edu_media = edu_media.reindex([e for e in edu_order if e in edu_media.index])
axes[1,1].bar(range(len(edu_media)), edu_media.values,
              color='#ffc107', edgecolor='white')
axes[1,1].set_xticks(range(len(edu_media)))
axes[1,1].set_xticklabels(edu_media.index, rotation=25, ha='right', fontsize=8)
axes[1,1].set_title('Retorno de la Educación')
axes[1,1].set_ylabel('Salario Medio (USD)')

plt.tight_layout()
plt.savefig('analisis_salarial_ecuador.png', dpi=150, bbox_inches='tight')
plt.show()
print("✅ Gráfico guardado")
```
