Contexto

Una empresa tecnológica de Quito monitorea el tiempo de carga (en segundos) de su aplicación web durante 25 días consecutivos. Como analista de datos, debes calcular las principales medidas estadísticas para presentar un informe al equipo directivo.

Dataset — Tiempos de carga diarios (segundos):

3.2, 2.8, 4.1, 3.5, 2.9, 4.8, 3.3, 2.7, 3.9, 4.2,
2.5, 3.6, 3.1, 4.4, 2.6, 3.8, 3.0, 4.6, 2.4, 3.7,
3.4, 2.9, 4.3, 3.2, 3.8

import numpy as np
import pandas as pd
from scipy import stats

tiempos = [3.2, 2.8, 4.1, 3.5, 2.9, 4.8, 3.3, 2.7, 3.9, 4.2,
           2.5, 3.6, 3.1, 4.4, 2.6, 3.8, 3.0, 4.6, 2.4, 3.7,
           3.4, 2.9, 4.3, 3.2, 3.8]

n = len(tiempos)
print(f"Total de observaciones: {n}")

# ─── PASO 2: Medidas de tendencia central ─────────────────────
media = # TU CÓDIGO AQUÍ (pista: np.mean)
mediana = # TU CÓDIGO AQUÍ (pista: np.median)

# La moda con scipy
resultado_moda = stats.mode(tiempos, keepdims=True)
moda = resultado_moda.mode[0]

print(f"\n📊 TENDENCIA CENTRAL")
print(f"  Media:   {media:.4f} seg")
print(f"  Mediana: {mediana:.4f} seg")
print(f"  Moda:    {moda:.1f} seg")

# ─── PASO 3: Medidas de dispersión ────────────────────────────
rango = # TU CÓDIGO AQUÍ (pista: max - min)
varianza = # TU CÓDIGO AQUÍ (pista: np.var)
desv_std = # TU CÓDIGO AQUÍ (pista: np.std)
cv = (desv_std / media) * 100

print(f"\n📏 DISPERSIÓN")
print(f"  Rango:           {rango:.4f}")
print(f"  Varianza:        {varianza:.4f}")
print(f"  Desv. Estándar:  {desv_std:.4f}")
print(f"  Coef. Variación: {cv:.2f}%")

# ─── PASO 4: Tabla de frecuencias ─────────────────────────────
serie = pd.Series(tiempos)
conteo = serie.value_counts().sort_index()

tabla = pd.DataFrame({
    'Valor': conteo.index,
    'fi': conteo.values,
    'hi': (conteo.values / n).round(4),
    'Fi': conteo.values.cumsum()
})

print("\n📋 TABLA DE FRECUENCIAS")
print(tabla.to_string(index=False))

# ─── PASO 5: Visualización ────────────────────────────────────
import matplotlib.pyplot as plt

fig, axes = plt.subplots(1, 2, figsize=(12, 4))

# Histograma
axes[0].hist(tiempos, bins=8, color='#0055a5', edgecolor='white', alpha=0.8)
axes[0].axvline(media, color='red', linestyle='--', linewidth=2, label=f'Media = {media:.2f}')
axes[0].axvline(mediana, color='green', linestyle='--', linewidth=2, label=f'Mediana = {mediana:.2f}')
axes[0].set_xlabel('Tiempo de carga (seg)')
axes[0].set_ylabel('Frecuencia')
axes[0].set_title('Distribución de Tiempos de Carga')
axes[0].legend()
axes[0].grid(axis='y', alpha=0.3)

# Boxplot
axes[1].boxplot(tiempos, vert=True, patch_artist=True,
                boxprops=dict(facecolor='#e8f4fd'),
                medianprops=dict(color='red', linewidth=2))
axes[1].set_ylabel('Tiempo de carga (seg)')
axes[1].set_title('Boxplot — Tiempos de Carga')
axes[1].grid(axis='y', alpha=0.3)

plt.suptitle('Análisis de Tiempos de Carga — Empresa TechEC', fontweight='bold')
plt.tight_layout()
plt.savefig('tiempos_carga.png', dpi=150, bbox_inches='tight')
plt.show()
print("✅ Gráfico guardado como 'tiempos_carga.png'")

# ─── PASO 1: Definir el dataset ──────────────────────────────
tiempos <- c(3.2, 2.8, 4.1, 3.5, 2.9, 4.8, 3.3, 2.7, 3.9, 4.2,
             2.5, 3.6, 3.1, 4.4, 2.6, 3.8, 3.0, 4.6, 2.4, 3.7,
             3.4, 2.9, 4.3, 3.2, 3.8)

n <- length(tiempos)
cat("Total de observaciones:", n, "\n")

# ─── PASO 2: Función para calcular la moda ───────────────────
calcular_moda <- function(x) {
  ux <- unique(x)
  ux[which.max(tabulate(match(x, ux)))]
}

# ─── PASO 3: Calcular medidas ────────────────────────────────
media   <- # TU CÓDIGO AQUÍ
mediana <- # TU CÓDIGO AQUÍ
moda    <- calcular_moda(tiempos)

rango    <- max(tiempos) - min(tiempos)

# Varianza poblacional (dividir por n, no n-1)
varianza <- sum((tiempos - media)^2) / n
desv_std <- sqrt(varianza)
cv       <- (desv_std / media) * 100

cat("\n📊 TENDENCIA CENTRAL\n")
cat(sprintf("  Media:   %.4f seg\n", media))
cat(sprintf("  Mediana: %.4f seg\n", mediana))
cat(sprintf("  Moda:    %.1f seg\n", moda))

cat("\n📏 DISPERSIÓN\n")
cat(sprintf("  Rango:           %.4f\n", rango))
cat(sprintf("  Varianza:        %.4f\n", varianza))
cat(sprintf("  Desv. Estándar:  %.4f\n", desv_std))
cat(sprintf("  Coef. Variación: %.2f%%\n", cv))

# ─── PASO 4: Tabla de frecuencias ────────────────────────────
freq_table <- as.data.frame(table(tiempos))
names(freq_table) <- c("Valor", "fi")
freq_table$hi <- round(as.numeric(freq_table$fi) / n, 4)
freq_table$Fi <- cumsum(as.numeric(freq_table$fi))

cat("\n📋 TABLA DE FRECUENCIAS\n")
print(freq_table)

# ─── PASO 5: Resumen con R ───────────────────────────────────
cat("\n📋 RESUMEN summary()\n")
print(summary(tiempos))

# ─── PASO 6: Visualización ───────────────────────────────────
par(mfrow = c(1, 2))

# Histograma con media y mediana
hist(tiempos,
     main = "Distribución de Tiempos de Carga",
     xlab = "Tiempo (seg)",
     col = "#0055a5",
     border = "white",
     breaks = 8)
abline(v = media,   col = "red",   lwd = 2, lty = 2)
abline(v = mediana, col = "green", lwd = 2, lty = 2)
legend("topright",
       legend = c(paste("Media =", round(media, 2)),
                  paste("Mediana =", round(mediana, 2))),
       col = c("red", "green"), lty = 2, lwd = 2)

# Boxplot
boxplot(tiempos,
        main = "Boxplot — Tiempos de Carga",
        ylab = "Tiempo (seg)",
        col = "#e8f4fd",
        border = "#003366")
abline(h = media, col = "red", lwd = 2, lty = 2)

par(mfrow = c(1, 1))
